// Profiling fixture only; instrumented throughput is not comparison evidence.
import 'dart:async';
import 'dart:convert';
import 'dart:developer';
import 'dart:io';
import 'package:flutter/material.dart';
import 'package:flutter/scheduler.dart';
import 'package:levit_flutter/levit_flutter.dart';
import 'file:///private/tmp/levit-performance-539f130-q8vb_rc4/benchmarks/lib/performance_monitor.dart';
final host = GlobalKey<_HostState>();
Future<void> nextFrame() => SchedulerBinding.instance.endOfFrame.timeout(const Duration(seconds: 5));
void main() {
  WidgetsFlutterBinding.ensureInitialized();
  Levit.enableAutoLinking();
  runApp(MaterialApp(home: _Host(key: host)));
  registerExtension('ext.levit.profile', (_, args) async {
    final enabled = Platform.environment['LEVIT_DIAGNOSTICS'] == 'on';
    final monitor = enabled ? (PerformanceMonitor()..enable()) : null;
    final version = 0.lx;
    var frames = 0;
    try {
      await nextFrame();
      await host.currentState!.mount(LWatch(() {
        final current = version.value;
        return ListView.builder(itemCount: 1000, itemExtent: 24,
          itemBuilder: (_, index) => Text('Row $index: $current'));
      }));
      for (var i = 0; i < 10; i++) { version.value++; await nextFrame(); }
      final elapsed = Stopwatch()..start();
      while (elapsed.elapsedMilliseconds < 5000) {
        version.value++;
        await nextFrame();
        frames++;
      }
      await monitor?.flush();
      if (monitor != null && monitor.dropped != 0) throw StateError('Diagnostic drops');
      return ServiceExtensionResponse.result(jsonEncode({'workload':'ui','diagnostics':enabled,'iterations':frames,'elapsedUs':elapsed.elapsedMicroseconds}));
    } finally {
      await host.currentState!.mount(const SizedBox.shrink());
      await Future<void>.sync(version.close);
      await monitor?.close();
    }
  });
  print('LEVIT_PROFILE_READY');
}
class _Host extends StatefulWidget {
  const _Host({super.key});
  @override State<_Host> createState() => _HostState();
}
class _HostState extends State<_Host> {
  Widget child = const SizedBox.shrink();
  Future<void> mount(Widget value) async { setState(() => child = value); await nextFrame(); }
  @override Widget build(BuildContext context) => Scaffold(body: child);
}
