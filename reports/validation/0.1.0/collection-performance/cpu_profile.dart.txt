import 'dart:async';
import 'dart:convert';
import 'dart:developer';
import 'dart:io';
import 'dart:isolate';
import 'package:levit_reactive/levit_reactive.dart';
import 'package:vm_service/vm_service_io.dart';

Future<void> main(List<String> args) async {
  final info = await Service.getInfo();
  final service = await vmServiceConnectUri(info.serverWebSocketUri.toString());
  final isolate = Service.getIsolateId(Isolate.current)!;
  final map = <int, int>{for (var i = 0; i < 128; i++) i: i}.lx;
  final list = List.generate(128, (i) => i).lx;
  var runs = 0;
  LxComputed<int> observe(int i) {
    final c = LxComputed(() {
      runs++;
      return args.first == 'shift' ? list[i] : map[i]!;
    });
    c.addListener(() {
      c.value;
    });
    return c;
  }

  final readers = args.first == 'shift'
      ? List.generate(128, observe)
      : <LxComputed<int>>[];
  Future<void> work() async {
    for (var i = 0; i < 256; i++) {
      if (args.first == 'shift') {
        list.insert(32, i);
        list.removeAt(32);
      } else {
        await observe(i % 128).close();
      }
    }
  }

  for (var i = 0; i < 10; i++) {
    await work();
  }
  await service.clearCpuSamples(isolate);
  await service.getAllocationProfile(isolate, reset: true, gc: true);
  final timer = Stopwatch()..start();
  var loops = 0;
  while (timer.elapsedMilliseconds < 2500) {
    await work();
    loops++;
  }
  final samples = await service.getCpuSamples(isolate, 0, 999999999999999);
  final alloc = await service.getAllocationProfile(isolate);
  final inclusive = <String, int>{}, self = <String, int>{};
  for (final sample in samples.samples!) {
    final names = sample.stack!
        .map((index) => samples.functions![index].function.name ?? 'unknown')
        .toList();
    if (names.isNotEmpty)
      self.update(names.first, (v) => v + 1, ifAbsent: () => 1);
    for (final n in names.toSet()) {
      inclusive.update(n, (v) => v + 1, ifAbsent: () => 1);
    }
  }
  Map<String, int> top(Map<String, int> m) => Map.fromEntries(
    (m.entries.toList()..sort((a, b) => b.value.compareTo(a.value))).take(50),
  );
  final allocations = <String, int>{};
  for (final s in alloc.members!) {
    allocations[s.classRef!.name!] = (s.instancesAccumulated ?? 0);
  }
  File(args[1]).writeAsStringSync(
    JsonEncoder.withIndent('  ').convert({
      'workload': args.first,
      'loops': loops,
      'evaluations': runs,
      'samples': samples.sampleCount,
      'inclusive': top(inclusive),
      'self': top(self),
      'allocations': top(allocations),
    }),
  );
  for (final c in readers) {
    await c.close();
  }
  await map.close();
  await list.close();
  await service.dispose();
}
